<?php
include_once "dbconnect.php";

$errorCategory = '';
$errorMerk = '';
$errorMemory = '';
$errorType = '';
$errorHd = '';
$errorPrijs = '';


$category = filter_input(INPUT_POST, 'category', FILTER_SANITIZE_SPECIAL_CHARS);

$merk = filter_input(INPUT_POST, 'merk', FILTER_SANITIZE_SPECIAL_CHARS);

$memory = filter_input(INPUT_POST, 'memory', FILTER_VALIDATE_FLOAT);

$type = filter_input(INPUT_POST, 'type', FILTER_SANITIZE_SPECIAL_CHARS);

$hd = filter_input(INPUT_POST, 'hd', FILTER_VALIDATE_FLOAT);

$prijs = filter_input(INPUT_POST, 'prijs', FILTER_VALIDATE_FLOAT);

$category = trim($category);

if (isset($_POST['insert'])) {
    if (empty($category)) {
        $errorCategory['category'] = 'category invullen';
    }
    if ($merk===false) {
        $errorMerk['merk'] = 'merk invullen';
    }
    if ($memory===false) {
        $errorMemory['memory'] = 'Memory invullen';
    }
    if ($type===false) {
        $errorType['type'] = 'type invullen';
    }
    if ($hd===false) {
        $errorHd['hd'] = 'hd invullen';
    }
    if ($prijs===false) {
        $errorPrijs['prijs'] = 'prijs invullen';
    }

    if ($errorCategory=='' && $errorMerk=='' && $errorMemory=='' && $errorType=='' && $errorHd=='' && $errorPrijs=='') {
        global $db;
        $query = $db->prepare('INSERT INTO laptops(category,merk,memory,type, hd, prijs) VALUES (:category,:merk,:memory,:type,:hd,:prijs)');
        $query->bindParam(':category', $category);
        $query->bindParam(':merk', $merk);
        $query->bindParam(':memory', $memory);
        $query->bindParam(':type', $type);
        $query->bindParam(':hd', $hd);
        $query->bindParam(':prijs', $prijs);
        $query->execute();
        header("location: read.php");
    }
}
